Googleスプレッドシート連携

Google Apps Script (GAS) のウェブアプリURLを設定すると、打刻時にGoogleスプレッドシートへリアルタイムで自動記録されます。 

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action || 'log';
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === 'monthly_sync') {
      var sheetName = contents.month + " 月次集計";
      var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
      sheet.clearContents();
      sheet.appendRow(["従業員ID", "氏名", "役職", "出勤日数", "総労働時間(時間)", "総残業時間(時間分)", "総残業時間(分)", "状態"]);
      var rows = contents.data || [];
      for (var i = 0; i < rows.length; i++) {
        var r = rows[i];
        sheet.appendRow([r.empId, r.name, r.role, r.daysWorked, r.totalWorkHours, r.overtimeFormatted, r.overtimeMinutes, r.status]);
      }
      return ContentService.createTextOutput(JSON.stringify({status:"success"})).setMimeType(ContentService.MimeType.JSON);
    }

    var sheet = ss.getSheetByName("打刻履歴") || ss.insertSheet("打刻履歴");
    if (sheet.getLastRow() === 0) sheet.appendRow(["日時", "従業員ID", "氏名", "役職", "打刻種別", "入力方法"]);
    sheet.appendRow([contents.timestamp || new Date().toLocaleString("ja-JP"), contents.empId||"", contents.name||"", contents.role||"", contents.type==="IN"?"出勤":"退勤", contents.method||"QR"]);
    return ContentService.createTextOutput(JSON.stringify({status:"success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:"error", message:err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
