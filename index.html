Google スプレッドシート連携

Google Apps Script (GAS) のウェブアプリURLを設定すると、打刻データや月次残業集計を自動的にスプレッドシートへリアルタイム同期できます。

【GASコード例】
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.appendRow([data.date, data.time, data.employeeId, data.name, data.role, data.type, data.method]);
  return ContentService.createTextOutput("OK");
}
